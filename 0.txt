	public static @NotNull RunCommandOutcome runCommand(          File                     folder
                                                  ,   @NotBlank String                   command
                                                              , long                     msTimeout
                                                              , Log                      log
                                                              , Function<String, String> handleCapturedOutput
                                                              , Function<String, String> showCapturedToUser
                                                              , List<String>             runInScriptIfWin
                                                              , String ...               args) throws IOException, InterruptedException, TimeoutException {
    
    assertTrue(handleCapturedOutput != null || showCapturedToUser == null
        , "handleCapturedOutput is", handleCapturedOutput, "but showCapturedToUser is", showCapturedToUser, ".", NL
                   , "If handleCapturedOutput is null then showCapturedToUser must be null too.");
    
    assertTrue(SystemUtils.IS_OS_WINDOWS || runInScriptIfWin == null, "The OS is not Windows, so the parameter 'runInScriptIfWin' must be null, instead it was:", NL2T, runInScriptIfWin);
    
    // Create the list containing the command in its first element and the args in the subsequent elements :
    
		final List<String> commandWithArgs = new ArrayList<>();
		
		commandWithArgs.add(assertNonBlank(command));
		
		if (args != null && args.length > ZERO_i) {
			
			commandWithArgs.addAll(Arrays.asList(args));
		}
    // If we are on Windows and the command starts with a pathless cmd.exe, prefix the command with the actual path to
    // that executable :
    
    String commandWithoutArgs = commandWithArgs.getFirst().stripLeading();
    
    final boolean usesWinCMD = SystemUtils.IS_OS_WINDOWS
                               && (commandWithoutArgs.equalsIgnoreCase(        "cmd")
                                || commandWithoutArgs.equalsIgnoreCase(        "cmd" + EXTENSION_SEPARATOR
                                                                                                   + "exe")
                                   || Strings.CI.startsWith(commandWithoutArgs, "cmd" + EXTENSION_SEPARATOR
                                                                                                   + "exe" + SPACE)
                                   || Strings.CI.startsWith(commandWithoutArgs, "cmd"         + SPACE));
    if (usesWinCMD) {
      
      final String cmdExe = getWinCOMSPEC();
      
      if (cmdExe != null) {
        
        commandWithoutArgs = cmdExe;
        
        commandWithArgs.set(ZERO_i, commandWithoutArgs);
      }
    }
    // Setup the Process Builder that will start the Process :
    
		final ProcessBuilder pb;
    
    final Process p;
    
    String wholeCommandLine = listToString(commandWithArgs, EMPTY, EMPTY
                                                           , EMPTY,    SPACE);
    Path tempScript = null;
    
    if (runInScriptIfWin != null) {
      
      // : Must create a script with all that needs to be executed, and execute that instead of directly the command :
      
      final List<String> scriptLines = new ArrayList<>(runInScriptIfWin);
      
      scriptLines.add(wholeCommandLine);
      
      scriptLines.add("EXIT /B %ERRORLEVEL%");
      
      tempScript = Files.createTempFile(OSUtilities.class.getSimpleName(), ".BAT");
      
      logIfLog(log, ONE_i, ZERO_i, false,"Creating temp script ", dq(tempScript.toFile().getAbsolutePath()), " containing the following ", I(scriptLines.size()), " lines:", NL2, listToString(scriptLines, EMPTY, TAB, EMPTY, NL));
      
      Files.write(tempScript, scriptLines);
      
      wholeCommandLine = tempScript.toAbsolutePath().toString();
      
      pb = new ProcessBuilder(wholeCommandLine);
    }
    else {
      
      // Must not create a script, must just execute the command directly :
      
      pb = new ProcessBuilder(commandWithArgs);
    }
    final StringBuilder processOutput;
    
    pb.directory(folder);
    
    if (handleCapturedOutput != null) {
      
      // : It is requested to capture the process' output :
      
      pb.redirectErrorStream(true);
      
      processOutput = new StringBuilder();
    }
    else {
      
      // : It is requested not to capture the process' output :
      
      assertNull(showCapturedToUser, "handleCapturedOutput =", handleCapturedOutput, "; showCapturedToUser =", showCapturedToUser);
      
      pb.redirectOutput(ProcessBuilder.Redirect.INHERIT);
      
      pb.redirectError( ProcessBuilder.Redirect.INHERIT);
      
      processOutput = null;
    }
    // Start the Process :
    
    logIfLog(log, ONE_i, "Starting OS process:" + NL2T + wholeCommandLine + NL2 + "with the following shell environment:" + NL2T + pb.environment().toString().replace(", ", NLT), ONE_i);
    
    p = pb.start();
    
    logIfLog(log, ONE_i, "OS process started, now waiting for completion, with timeout of " + msTimeout + " ms ...", TWO_i);
    
    if (handleCapturedOutput != null) {
      
      // : It is requested to capture the process' output :
      
      final Charset readerCharset = StandardCharsets.UTF_8; // Charset.defaultCharset()
      
      logIfLog(log, TWO_i, DASH80 + NL + "BEGINNING OF OS PROCESS' OUTPUT." + NL2
                                                    + "It will be captured using charset " + readerCharset.name() + NL
                                                    + DASH80, TWO_i);
      
      try (final BufferedReader reader = new BufferedReader(new InputStreamReader(p.getInputStream()
                                                                                    , readerCharset))) {
        String line;
        
        while ((line = reader.readLine()) != null) {
          
          line = handleCapturedOutput.apply(line);
          
          processOutput.append(line);
          
          logIfLog(log,  line); // : To the log if given.
          
          if (showCapturedToUser != null) {
            
            // : It is requested to still show to the user the process' captured output :
            
            showCapturedToUser.apply(line);    // : To the user (e.g. console, or other).
          }
        }
      }
    }
    assertSameNullness(runInScriptIfWin, tempScript);
    
    // Wait with timeout :
    
    final boolean timedOut = ! p.waitFor(msTimeout, TimeUnit.MILLISECONDS);
    
    logIfLog(log, TWO_i, DASH80 + NL + "END OF OS PROCESS' OUTPUT" + NL + DASH80, TWO_i);
    
    if (tempScript != null) {
      
      // : Mark the temporary script for deletion at shutdown no matter what :
      
      tempScript.toFile().deleteOnExit();
    }
    if (! timedOut) {
      
      // : Delete the possible temporary script right away if its execution succeeded (no timeout and no exception).
      
      if (tempScript != null) {
        
        Files.delete(tempScript);
      }
    }
    else {
      
      // : The wait exceeded the timeout, so throw TimeoutException :
      
      logIfLog(log, ZERO_i, "The OS process duration exceeded the timeout (" + msTimeout + " ms). Now destroying the process forcibly ...", ZERO_i);
      
      p.destroyForcibly();
      
      final TimeoutException e = new TimeoutException("Timeout of " + msTimeout + " ms exceeded by the following command :" + NL2T2 + wholeCommandLine);
      
      logIfLog(log, ONE_i, e, ZERO_i);
      
      throw e;
    }
    return new RunCommandOutcome(wholeCommandLine, p.exitValue(), pb.directory()
                 , processOutput != null ? processOutput.toString() : null);
	}
