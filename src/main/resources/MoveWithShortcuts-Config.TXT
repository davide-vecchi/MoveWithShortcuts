# MoveWithShortcuts configuration file.
#
# When running from the IDE, this file is read from src/main/resources.
# For the deployed JAR, place a copy of this file next to MoveWithShortcuts.jar.

#
# ShortcutsUpdater : the IShortcutsUpdater implementation to use.
# Accepted values  : ps-scripts-multi.
# If this file or this parameter is missing, the default implementation is used.
#
ShortcutsUpdater = ps-scripts-multi

#
# CreateContextMenu : whether to create or update the Windows Explorer right-click context-menu integration at startup.
# Accepted values : true, false (yes / no are accepted too). Default : true.
#
CreateContextMenu = true

#
# Console : whether to start in console mode instead of the default GUI mode.
# Accepted values : true, false (yes / no are accepted too). Default : false.
#
# true / yes makes console mode the default; a -gui command line arg overrides it.
#
Console = false

#
# DefaultSearchPath : the default folder to scan for shortcuts when -search / -default-search-path is not given.
# It is used to prefill the GUI search field and as the console prompt default. If absent, the user's home folder is
# used. A value that is blank, missing, not a folder or invalid falls back to the user's home folder and produces a
# warning at startup.
#
DefaultSearchPath = H:\Users\Davide\Davide
